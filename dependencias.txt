REM ─────────────────────────────────────────────────────────────────
REM  Dependencias de IntelliHelp — Motor de Campañas
REM  Ejecutar este archivo (o copiar-pegar las líneas) en una PC nueva
REM  antes de correr "py interfaz.py".
REM ─────────────────────────────────────────────────────────────────

REM 1) Actualizar pip
python.exe -m pip install --upgrade pip

REM 2) Flet (framework UI)
pip install flet

REM 3) html2image — renderiza el HTML del correo como imagen usando
REM    Chrome headless. Esto permite que en la pestaña "Conversación"
REM    cada correo se vea EXACTAMENTE igual a Gmail (logos, gradientes,
REM    firmas con imágenes incrustadas, SVG, todo).
REM    Requiere Chrome o Chromium instalado (Edge NO sirve).
pip install html2image

REM 4) plyer — notificaciones nativas del SO (toast Windows 10/11) cuando
REM    llega una respuesta nueva al hilo. Si no está, la app cae a un beep.
pip install plyer

REM 5) (Opcional) flet-webview — sólo funciona en mobile/web.
REM    En desktop Windows NO se usa, pero no rompe nada si está instalado.
pip install flet-webview
